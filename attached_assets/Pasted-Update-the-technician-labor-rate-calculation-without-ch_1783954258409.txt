Preserve that behavior exactly.

effectiveHighestPaidRate = calculatorOverride ?? settingsHighestPaidFieldTechnician

laborRate = effectiveHighestPaidRate * 3.5

annualBillableHours = annualPaidHours * billableUtilization

burdenedHourlyWage = effectiveHighestPaidRate * (1 + payrollBurdenPercent)

truckCostPerBillableHour = annualTruckFieldCost / annualBillableHours

directCostPerBillableHour = laborCostPerBillableHour + truckCostPerBillableHour

recoveryMargin = 1 - targetOverheadPercent - targetOperatingProfitPercent

rawLaborRate = directCostPerBillableHour / recoveryMargin

recommendedLaborRate = Math.ceil(rawLaborRate / roundingIncrement) * roundingIncrement

impliedWageMultiplier = recommendedLaborRate / effectiveHighestPaidRate
